{{- /* The repository's files and directories (path → true), read from .generated/repo-files.txt, which `make docs-assets` writes with `git ls-files`. Hugo cannot stat files outside docs-site/, so this list is how the link hook tells a link to a repository file from a broken one. Returns false when the list is missing (a plain `hugo server`); links are then not checked against it. */ -}} {{- $files := false -}} {{- $list := ".generated/repo-files.txt" -}} {{- if os.FileExists $list -}} {{- $s := newScratch -}} {{- $s.Set "files" dict -}} {{- range split (os.ReadFile $list) "\n" -}} {{- with . -}} {{- $s.SetInMap "files" . true -}} {{- $dir := path.Dir . -}} {{- range seq 8 -}} {{- if ne $dir "." -}} {{- $s.SetInMap "files" $dir true -}} {{- $dir = path.Dir $dir -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $files = $s.Get "files" -}} {{- end -}} {{- return $files -}}
