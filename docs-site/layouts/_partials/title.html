{{- /* The title of a page. Pages of docs-site/content set `title` in front matter; pages mounted from docs/ have no front matter and take the text of their first level-1 heading, as GitHub shows them. */ -}} {{- $title := "" -}} {{- with .Params.title -}} {{- $title = . -}} {{- else -}} {{- with findRESubmatch `(?m)^#[ \t]+(.+?)[ \t#]*$` .RawContent 1 -}} {{- $title = index (index . 0) 1 | $.RenderString | plainify | htmlUnescape -}} {{- else -}} {{- $title = .Title -}} {{- end -}} {{- end -}} {{- return $title -}}
